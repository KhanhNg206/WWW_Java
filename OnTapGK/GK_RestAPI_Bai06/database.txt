    TENDANHMUC VARCHAR(100) NOT NULL,
    NGUOIQUANLY VARCHAR(100),

    MATT VARCHAR(20) PRIMARY KEY,
    TIEUDE VARCHAR(255) NOT NULL,
    NOIDUNGTT TEXT,
    LIENKET VARCHAR(500),
    MADM INT NOT NULL,
    CONSTRAINT FK_TINTUC_DANHMUC
        FOREIGN KEY (MADM) REFERENCES DANHMUC(MADM)
);
